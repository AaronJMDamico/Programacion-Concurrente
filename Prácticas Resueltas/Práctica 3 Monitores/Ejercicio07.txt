Se debe simular una maratón con C corredores donde en la llegada hay UNA máquina
expendedora de agua con capacidad para 20 botellas. Además, existe un repositor encargado
de reponer las botellas de la máquina. Cuando los C corredores han llegado al inicio, comienza
la carrera. Cuando un corredor termina la carrera, se dirige a la máquina expendedora, espera
su turno (respetando el orden de llegada), saca una botella y se retira. Si encuentra la máquina
sin botellas, le avisa al repositor para que cargue nuevamente la máquina con 20 botellas;
espera a que se haga la recarga; saca una botella y se retira.
Nota: mientras se reponen las botellas, se debe permitir que otros corredores se encolen.

Process Corredor [id: 1..C] {
	Carrera.empezar()
	Fila.llegar
	Expendedora.sacarBotella()
	Fila.irse
}

Process Reponedor {
	while (true) {
		Expendedora.reponer()
	}
}

Monitor Fila {
	
	bool libre = true
	int esperando = 0
	cond turno
	
	Procedure llegar() {
		if (not libre) {
			esperando++
			wait(turno)
		} else {
			libre = false
		}
	}
	
	Procedure retirar() {
		if (esperando > 0) {
			esperando--
			signal(turno)
		} else {
			libre = true
		}
	}
}

Monitor Expendedora {

	capacidad = 20
	cond noHay
	cond hayBotellas

	Procedure sacarBotella() {
		if (capacidad == 0) {
			signal(noHay)
			wait(hayBotellas)
		}
		// Saca la botella
		capacidad--
	}

	Procedure  reponer() {
		if (capacidad > 0) {
			wait (noHay)
		}
		capacidad = 100
		signal(hayBotellas)
	}
}

Monitor Carrera {
	int llegados = 0
	cond espera

	Procedure empezar() {
		llegados++
		if (llegados == C) {
			signalAll(espera)
		} else {
			wait(espera)
		}
	}
}
