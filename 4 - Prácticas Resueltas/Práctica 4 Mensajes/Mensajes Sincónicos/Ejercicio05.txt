5) En un estadio de fútbol hay una máquina expendedora de gaseosas que debe ser usada por
E Espectadores de acuerdo con el orden de llegada. Cuando el espectador accede a la
máquina en su turno, usa la máquina y luego se retira para dejar al siguiente. Nota: cada
espectador usa la máquina sólo una vez.

Process Espectador [id: 0..E-1] {
	Admin!pedirAcceso(id)
	Maquina?esperarAcceso()
	Maquina!liberarAcceso()
}

Process Maquina {
	int idAux = ?

	while (true) {
		Admin!libre()
		Admin?recibirPersona(idAux)
		Espectador[idAux]!esperarAcceso()
		Espectador[idAux]?liberarAcceso()
	}
}

Process Admin {
	int idAux = ?
	queue Fila(int)

	while (true) {
		do Espectador[*]?pedirAcceso(idAux) ->
			push(Fila, idAux)
		□ not empty(Fila); Maquina?libre() ->
			pop(Fila, idAux)
			Maquina!recibirPersona(idAux)
		od
	}
}

process Admin
{
	queue Fila(int);
    bool maquinaLibre = true;
    int cantEspectadores = 0;
    int esperando = 0;
	int idAux = ?;
	int idSig = ?;

    do cantEspectadores < E; Espectador[*]?pedirAcceso(idAux) ->
        if(maquinaLibre) 
		{
            //Da acceso a la maquina al siguiente
            Espectador[idAux]!esperarAcceso();
            maquinaLibre = false;
        } 
		else {
            //Si estan usando la maquina, espera en la cola
            push(Fila, idAux);
            esperando++;
        }
        cantEspectadores++;
    □ maquinaLibre = false; Espectador[*]?liberarAcceso() ->
        if(esperando > 0) 
		{
            //Despierta al siguiente
            pop(Fila, idSig);
            Espectador[idSig]!esperarAcceso();
            esperando--;
        } 
		else {
            maquinaLibre = true;
        }
    od
}
