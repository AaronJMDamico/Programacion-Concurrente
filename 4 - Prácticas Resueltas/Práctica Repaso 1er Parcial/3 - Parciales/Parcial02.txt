3) 

Resolver con MONITORES el siguiente problema. En una acopiadora de cereales hay 2 empleados, uno para 
atender a los camiones de maíz y otro para los de girasol. Hay 30 camiones que llegan para descargar su carga
(15 de maíz y 15 de girasol), cuando el camión llega espera hasta que el empleado correspondiente l avise que puede
descargar el cereal. Cada empleado hace descargar los camiones que le corresponden de a uno a la vez y de acuerdo
con orden de llegada. Nota: maximizar concurrencia; el camión sabe que tipo de cereal lleva; todos los procesos
deben terminar.

Process Empleado [id: 1, 2] {
	int cant = 0
	bool termine = false

	while (not termine) {
		Acopiadora[id].darPaso()
		cant++
		if (cant == 15) {
			termine = true
		}
	}
}

Process Camiones [id: 0..29] {
	int tipo = random(1, 2)
	
	Acopiadora[tipo].llegar()
	Descargar()
	Acopiadora[tipo].irse()
}

Monitor Acopiadora[1, 2] {
	int camiones = 0
	cond noHay
	cond espera
	cond descarga

	Procedure llegar() {
		camiones++
		signal(noHay)
		wait(espera)
	}
	
	Procedure darPaso() {
		if (camiones == 0) {
			wait(noHay)
		}
		camiones--
		signal(espera)
		wait(descarga)
	}
	
	Procedure irse() {
		signal(descarga)
	}
}
